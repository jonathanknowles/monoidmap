-- | Constructs a 'NonNullMap' from a list of key-value pairs, combining the
--   values for each key with '(<>)' in the order that they appear, and
--   removing all 'null' values.
--
-- This function selects which of the implementations in
-- "Data.MonoidMap.Internal.NonNullMap.FromList" is used by the public
-- @fromList@ function of the @monoidmap@ package.
--
fromList :: (Ord k, MonoidNull v) => [(k, v)] -> NonNullMap k v
fromList :: forall k v. (Ord k, MonoidNull v) => [(k, v)] -> NonNullMap k v
fromList = [(k, v)] -> NonNullMap k v
forall k v. (Ord k, MonoidNull v) => [(k, v)] -> NonNullMap k v
fromList_flip_filter_if_any_null
{-# INLINE fromList #-}

-- | Constructs a 'NonNullMap' from an ordinary 'Map', removing all
--   'null' values.
--
-- This function selects which of the implementations in
-- "Data.MonoidMap.Internal.NonNullMap.FromMap" is used by the public @fromMap@
-- function of the @monoidmap@ package.
--
fromMap :: MonoidNull v => Map k v -> NonNullMap k v
fromMap :: forall v k. MonoidNull v => Map k v -> NonNullMap k v
fromMap = Map k v -> NonNullMap k v
forall v k. MonoidNull v => Map k v -> NonNullMap k v

{-# INLINE fromMap #-}

-- | Constructs a 'NonNullMap' from an ordinary 'Map', applying the given

fromMapWith = (v1 -> v2) -> Map k v1 -> NonNullMap k v2
forall v2 v1 k.
MonoidNull v2 =>

{-# INLINE fromMapWith #-}
